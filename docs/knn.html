Eliminación de datos nulos

df = df[df['survived'].notna()]
df['survived'] = df['survived'].astype(int)

Separación

X = df.drop(columns='survived')
y = df['survived']

Identificación de Features

cat_cols = ['sex','embarked']
num_cols = ['age','sibsp','parch','pclass']

PipeLine

Preproecsamiento

Categorilas

cat_pipeline = Pipeline([
    ('imputacion_cat',SimpleImputer(strategy='most_frequent')), 
    ('encodage_cat',OneHotEncoder(handle_unknown='ignore')),
])

Numerical PipeLines

num_pipeline = Pipeline([
    ('imputacion_num',SimpleImputer(strategy='most_frequent')), 
    ('escalado_num',StandardScaler()),
])

Preprocessor

preprocessor = ColumnTransformer([
    ('cat',cat_pipeline,cat_cols),
    ('num',num_pipeline,num_cols)
])

Pipeline de Clasificador

pipeline = Pipeline([
    ('preprocessor',preprocessor),
    ('clasificador',KNeighborsClassifier())
])

Train_test_split

X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size= 0.2,
    stratify=y,
    random_state=42,
)
